    ```python
    def _fetch_via_csv(csv_path: Path = Path("data/rcp_2024_general.csv")) -> list[dict]:
        """Laedt RCP-Daten aus einer heruntergeladenen CSV-Datei.

        Erwartet eine CSV mit Datum, Trump-Prozent und Harris-Prozent Spalten.
        Flexibler Spaltenname-Match fuer gaengige RCP-Export-Formate.

        Args:
            csv_path: Pfad zur CSV-Datei. Standard: data/rcp_2024_general.csv.

        Returns:
            Liste von Dicts mit Schluesseln date (YYYY-MM-DD), trump_pct, harris_pct.
            Leere Liste wenn Datei nicht existiert oder nicht lesbar.
        """
        if not Path(csv_path).exists():
            logger.warning("RCP CSV nicht gefunden: %s", csv_path)
            return []

        rows: list[dict] = []
        try:
            with open(csv_path, newline="", encoding="utf-8-sig") as f:
                reader = csv.DictReader(f)
                headers = [h.strip().lower() for h in (reader.fieldnames or [])]

                # Detect Trump and Harris columns by fuzzy match
                trump_col = next(
                    (h for h in headers if "trump" in h.lower()), None
                )
                harris_col = next(
                    (h for h in headers if "harris" in h.lower()), None
                )
                date_col = next(
                    (h for h in headers if "date" in h.lower()), None
                )

                if not (trump_col and harris_col and date_col):
                    logger.warning(
                        "RCP CSV: Erwartete Spalten nicht gefunden. Headers: %s", headers
                    )
                    return []

                for raw_row in reader:
                    # Normalize header keys to lowercase-stripped for lookup
                    row = {k.strip().lower(): v.strip() for k, v in raw_row.items()}
                    try:
                        # Parse date — handle MM/DD/YYYY and YYYY-MM-DD
                        raw_date = row[date_col]
                        if "/" in raw_date:
                            dt = datetime.strptime(raw_date, "%m/%d/%Y")
                        else:
                            dt = datetime.strptime(raw_date[:10], "%Y-%m-%d")
                        date_str = dt.strftime("%Y-%m-%d")

                        trump_pct = float(row[trump_col])
                        harris_pct = float(row[harris_col])

                        if trump_pct > 0.0 and harris_pct > 0.0:
                            rows.append({
                                "date": date_str,
                                "trump_pct": trump_pct,
                                "harris_pct": harris_pct,
                            })
                    except (ValueError, KeyError) as exc:
                        logger.debug("RCP CSV Zeile uebersprungen: %s", exc)
                        continue

        except Exception as exc:
            logger.warning("RCP CSV Lesefehler: %s", exc)
            return []

        logger.info("RCP CSV: %d Zeilen gelesen aus %s", len(rows), csv_path)
        return rows
    ```
